cd $PBS_O_WORKDIR;

## define singularity container to use
container=/home/svu/e1036284/multiScaleFusionsKD3A.sif
## fusion_method: weighted_projection

	--pwd /multiscale_KD3A \
	--bind /scratch/e1036284/I3D-feature-pretrain/:/multiscale_KD3A/data/I3D-feature-pretrain \
	--bind /home/svu/e1036284/EpicKitchensMultiScaleWeightedProjection.yaml:/multiscale_KD3A/config/EpicKitchensMultiScale.yaml \

	--target-domain P22 --source-domains P08 P01 \
	--data-parallel \
	--train-time 13  \

        --pwd /multiscale_KD3A \
